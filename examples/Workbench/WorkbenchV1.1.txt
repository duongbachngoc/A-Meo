A Meo Workbench module -- developer source package (V1.1)

This bundle contains the complete Workbench module source under
modules/workbench/. Materialize it with the build_project.py sitting next
to this file:

    python build_project.py [--project-root PATH]

Workbench is an autonomous "work" module for A Meo: given a goal and a
workspace folder, it runs an OBSERVE -> DECIDE -> ACT -> VERIFY loop
(read / search / patch files, run bounded commands) until it reports
COMPLETE, WAIT_OWNER, or FAILED. It integrates with A Meo strictly through
the Public Module Contract v1: a process-isolated child process, a
JSON-lines stdin/stdout protocol, and the `invoke()` Host API allowlist.
Workbench is USED BY A Meo; it does not become part of A Meo.

The only Core import Workbench depends on is
`from src.core.hub.contracts import BaseCapability, HubResponse`
(the one stable public type, provided by the A Meo Host at runtime -- see
the Public Module Contract). Everything else under modules/workbench/ is
self-contained.

def _make_stub_invoke():
    """Fake `conversation.chat`: first call writes note.txt, second call
    signals complete. Deterministic and needs no real AI provider/API key

    call_count = {"n": 0}

    async def _stub_invoke(capability: str, action: str, payload: dict):
        from src.core.hub.contracts import HubResponse

        if capability == "conversation" and action == "chat":

            if call_count["n"] == 1:
                decision = {
                    "action": "write_file",
                    "params": {"path": "note.txt", "content": "hello from workbench\n"},
                    "reasoning": "create the file the goal asked for",

async def _main() -> None:
    with tempfile.TemporaryDirectory() as tmp_data, tempfile.TemporaryDirectory() as tmp_workspace:
        cap = boot(invoke=_make_stub_invoke(), module_data_root=Path(tmp_data))
        assert cap.capability_name == "workbench"
        assert cap.health_check() is True

        start_resp = await cap.execute("handle_turn", {
            "text": f"start {tmp_workspace} :: tao file note.txt",
            "user_id": "local-dev",
        })
        print("start:", start_resp.data)
        assert start_resp.data["consumed"] is True

        note_path = Path(tmp_workspace) / "note.txt"

        for _ in range(100):
            await asyncio.sleep(0.05)
            status_resp = await cap.execute("handle_turn", {"text": "status", "user_id": "local-dev"})

        assert "HOAN THANH" in final_response, f"expected COMPLETE, got: {final_response}"

        print("note.txt content:", note_path.read_text(encoding="utf-8").strip())

"""WorkbenchCapability -- the module's single BaseCapability, and the only
place `handle_turn` logic lives. Per the frozen runtime rule, `execute()`
NEVER awaits the agent loop: it only starts/resumes it as a background
asyncio.Task and returns immediately (Contract muc 6.2's 60s handle_turn
timeout).

Agent-first rewrite (owner directive): this file used to run a second AI
call just to classify whether an incoming owner message was a "pivot"
(`_is_goal_pivot`/`_is_running_pivot`/`_classify_pivot`) before deciding
whether to touch `session.goal`. That machinery is deleted. There is no
Python "is this a pivot / is this an answer / is this approval" judgement
anywhere in this file anymore: ANY owner text arriving while a session is
RUNNING or WAITING is simply appended to the real conversation transcript
and the loop is (re)spawned. The model reads the full transcript next turn
and decides for itself what the new text means -- that decision is no
longer this file's job."""

_IDLE_LIKE = (STATUS_COMPLETE, STATUS_FAILED, STATUS_CANCELLED, STATUS_IDLE)

_NEEDS_WORKSPACE_MESSAGE = (

    "Please select a 'Source Folder' in A Meo first, or use the manual syntax:\n"
    "start <workspace_path> :: <goal>"

_NO_USABLE_FOLDER_MESSAGE = (

_SNAPSHOT_MAX_ENTRIES = 5000


def _snapshot_existing_paths(workspace_root) -> list:
    """Every relative path (files AND directories) already present under
    workspace_root at the moment a session starts -- becomes
    WorkSession.protected_paths (see file_tools.ProtectedSourceFileError).
    Best-effort: an OSError partway through a walk (permissions, a
    transient lock) must never block starting the session -- it just means
    whatever was already collected is what gets protected; never silently
    treat a scan failure as "nothing to protect"."""
    from pathlib import Path
    root = Path(workspace_root)
    paths = []
    try:
        for p in root.rglob("*"):
            try:
                paths.append(p.relative_to(root).as_posix())
            except ValueError:
                continue
            if len(paths) >= _SNAPSHOT_MAX_ENTRIES:
                break
    except OSError:
        pass
    return paths


def _multiple_folders_message(goal: str, paths: list) -> str:
    lines = [
        "The current Source Folder has more than 1 usable folder, Workbench "
        "will not guess which one to pick. Please choose one path using the syntax:",
    ]
    lines.extend(f"- {p}" for p in paths)
    lines.append(f"start <path_above> :: {goal}")
    return "\n".join(lines)


# Owner-facing chat UX: normal chat replies never show session_id/workspace
# path/step count/tool names/agent-loop wording -- that detail stays in the
# persisted session JSON (steps/conversation are still written there in
# full; this file just stops echoing them into the chat bubble). While a
# WorkSession is RUNNING, every reply is this one fixed sentinel so the
# Electron renderer can render it as the existing pulsing ".typing"
# indicator (app.js) instead of literal text, and keep polling until a
# real terminal/waiting reply arrives.
_WORKING_INDICATOR = "..."

# Small, fully deterministic map from the FIXED set of internal
# result_summary strings agent_loop.py itself produces (Python-triggered
# terminal failures, where no model-authored text exists to surface) to a
# short, honest, non-technical reason -- not a general error-translation
# engine (the handful of FAILED causes this module can produce are
# enumerable).
_FAILED_REASON_MAP = [
    ("vuot qua so buoc", "tried several approaches but did not finish within the allowed limit"),
    ("Loi goi AI", "could not connect to the AI"),
    ("AI invoke that bai", "could not connect to the AI"),
    ("AI lien tuc tra loi sai dinh dang", "the AI replied in the wrong format multiple times"),
    ("Qua nhieu buoc lien tiep that bai", "ran into repeated errors while performing the action"),
]


def _owner_failed_reason(result_summary: Optional[str]) -> str:
    text = result_summary or ""
    for needle, reason in _FAILED_REASON_MAP:
        if needle in text:
            return reason
    return "ran into an error while carrying out the request"


def _last_assistant_text(session: WorkSession) -> Optional[str]:
    for turn in reversed(session.conversation):
        if turn.get("role") == "assistant":
            return turn.get("text")
    return None


def _current_activity_text(session: WorkSession) -> str:
    """Owner-facing text while RUNNING -- derived ONLY from the most
    recent real `tool_call` step already recorded in `session.steps`
    (tool name + its own params), via `agent_loop.describe_tool_activity`.
    Pure observability of a physical event that already happened, never a
    model call and never an inference of "what the AI is thinking" --
    replaces the old static `_WORKING_INDICATOR` placeholder (and the
    removed `progress` decision, which used to cost a full extra model
    call to say the same kind of thing, yet was never even surfaced here
    before this fix). Falls back to the static indicator only when no
    tool_call has happened yet (session just started).

    Always PREFIXED with `_WORKING_INDICATOR` itself (never returned bare)
    -- app.js's poll loop (electron/renderer/app.js,
    WORKBENCH_WORKING_INDICATOR) recognizes "still RUNNING, keep polling"
    by matching that exact sentinel; switching to a `startswith` check
    there (instead of `===`) lets the real activity text ride along
    without breaking that existing contract or adding a second field to
    the handle_turn response shape."""
    for step in reversed(session.steps):
        decision = step.get("decision") or {}
        if decision.get("type") == "tool_call":
            activity = agent_loop.describe_tool_activity(decision.get("tool"), decision.get("params"))
            return f"{_WORKING_INDICATOR} {activity}"
    return _WORKING_INDICATOR


def _status_message(session: Optional[WorkSession]) -> str:
    """Concise, owner-facing state -- no session_id/path/step count/tool
    names/agent-loop wording. This is also what a plain 'status' turn
    returns; full technical detail (every step, tool call, raw AI
    reasoning) remains available only in the persisted session JSON, not
    in normal chat. For WAITING/COMPLETE, this surfaces the model's OWN
    composed text verbatim -- no Python-templated phrasing -- since under
    the agent-first rewrite the model owns how it talks to the owner."""
    if session is None:
        return "No work session yet. Please describe the task to do, or select a 'Source Folder' first."
    if session.status == STATUS_RUNNING:
        return _current_activity_text(session)
    if session.status == STATUS_WAITING:
        return _last_assistant_text(session) or "Waiting for your confirmation on a choice before continuing."
    if session.status == STATUS_COMPLETE:
        return session.result_summary or "Done."
    if session.status == STATUS_FAILED:
        return f"Could not fix it: {_owner_failed_reason(session.result_summary)}."
    if session.status == STATUS_CANCELLED:
        return "The task was cancelled."
    return "Ready."


class WorkbenchCapability(BaseCapability):
    def __init__(self, store: SessionStore, invoke: InvokeFn):
        self._store = store
        self._invoke = invoke
        self._task: Optional[asyncio.Task] = None
        self._task_session_id: Optional[str] = None

    @property
    def capability_name(self) -> str:
        return MODULE_ID

    def health_check(self) -> bool:
        return True

    async def execute(self, action: str, payload: Dict[str, Any]) -> HubResponse:
        if action != "handle_turn":
            return HubResponse(success=False, error=f"Unsupported action '{action}'.")
        text = (payload.get("text") or "").strip()
        try:
            consumed, response = await self._route(text)
        except Exception as exc:  # noqa: BLE001 -- Contract muc 7.4: never raise a raw exception out
            return HubResponse(success=True, data={
                "consumed": True,
                "response": f"Workbench encountered an internal error: {exc}",
            })
        return HubResponse(success=True, data={"consumed": consumed, "response": response})

    def _spawn_task_if_needed(self, session_id: str) -> None:
        if self._task is not None and not self._task.done() and self._task_session_id == session_id:
            return  # already running for this session -- idempotent start/resume
        self._task = asyncio.create_task(agent_loop.run_session(session_id, self._store, self._invoke))
        self._task_session_id = session_id

    def _start_session(self, workspace_root, goal: str) -> WorkSession:
        """Single place a new WorkSession is created and marked RUNNING.
        Creates the session already-RUNNING in ONE atomic save
        (SessionStore.new_session's `status=` kwarg) instead of
        create-then-mutate-then-save-again -- a confirmed real bug this
        replaces was exactly that second save failing transiently and
        leaving the background task never spawned."""
        session = self._store.new_session(
            str(workspace_root), goal,
            status=STATUS_RUNNING,
            protected_paths=_snapshot_existing_paths(workspace_root),
        )
        self._spawn_task_if_needed(session.session_id)
        return session

    async def _route(self, text: str) -> Tuple[bool, str]:
        session = self._store.get_current()
        lowered = text.lower()

        # 'status' always answers, whatever state the session is in (or
        # even if there is none yet) -- this is the "get current
        # status/result" handle_turn responsibility from the frozen
        # runtime rule, and it must work after the background task has
        # already reached a terminal state, not just while RUNNING.
        if lowered == "status":
            if session is not None and session.status == STATUS_RUNNING:
                # (Re)spawn if this RUNNING session has no live task -- this
                # is what makes a fresh process resume a persisted session
                # after restart/deactivate-reactivate, even via a plain
                # status check, without needing a dedicated 'resume' verb.
                self._spawn_task_if_needed(session.session_id)
            return True, _status_message(session)

        if lowered == "cancel" and session is not None and session.status not in _IDLE_LIKE:
            if self._task is not None and not self._task.done() and self._task_session_id == session.session_id:
                self._task.cancel()
            session.status = STATUS_CANCELLED
            self._store.save(session)
            return True, "The task was cancelled."

        if session is None or session.status in _IDLE_LIKE:
            if lowered.startswith("start "):
                return True, self._handle_start(text[len("start "):])
            if not text:
                return False, ""  # nothing to respond to
            # Workbench is Active but idle: any other non-empty text is a
            # natural-language task attempt, not silence. Resolve the
            # owner's already-selected "Thu muc nguon" through the public
            # Host API and auto-start there.
            return await self._handle_natural_language_start(text)

        # RUNNING or WAITING: no classification of what the new text MEANS
        # -- just append it to the real conversation transcript and
        # (re)spawn the loop. The model reads the full transcript next turn
        # and decides for itself whether this is an answer, a correction, a
        # brand-new instruction, or approval of a pending confirmation.
        if session.status in (STATUS_RUNNING, STATUS_WAITING):
            if text:
                current_root = await self._current_active_workspace_root()
                if current_root is not None and str(current_root) != session.workspace_root:
                    if self._task is not None and not self._task.done() and self._task_session_id == session.session_id:
                        self._task.cancel()
                    session.status = STATUS_CANCELLED
                    self._store.save(session)
                    return await self._handle_natural_language_start(text)
                session.conversation.append({"role": "owner", "text": text})
            elif session.status != STATUS_RUNNING:
                # Nothing new to say and nothing already running -- no-op.
                return False, ""
            session.status = STATUS_RUNNING
            self._store.save(session)
            self._spawn_task_if_needed(session.session_id)
            return True, _WORKING_INDICATOR

        return True, _status_message(session)

    async def _current_active_workspace_root(self):
        """Resolves the owner's CURRENTLY selected "Thu muc nguon" to a
        single absolute workspace root, exactly like `_handle_natural_
        language_start` does when starting a fresh session -- returns None
        (never raises) on ANY ambiguity/missing-selection/Host-API failure,
        so callers fail closed to their prior behavior instead of blocking
        an ordinary turn on uncertainty."""
        try:
            ctx_resp = await self._invoke("knowledge_source_manager", "get_runtime_context", {})
        except Exception:  # noqa: BLE001
            return None
        if not getattr(ctx_resp, "success", False):
            return None
        active_source_id = ((ctx_resp.data or {}).get("runtime_context") or {}).get("active_source_id")
        if not active_source_id:
            return None
        try:
            folders_resp = await self._invoke("knowledge_source_manager", "list_folders", {"source_id": active_source_id})
        except Exception:  # noqa: BLE001
            return None
        if not getattr(folders_resp, "success", False):
            return None
        folders = (folders_resp.data or {}).get("folders") or []
        usable_paths = [f["path"] for f in folders if f.get("path")]
        if len(usable_paths) != 1:
            return None
        try:
            return resolve_workspace_root(usable_paths[0])
        except WorkspaceError:
            return None

    def _handle_start(self, rest: str) -> str:
        if "::" not in rest:
            return "Syntax: start <workspace_path> :: <goal>"
        raw_path, goal = rest.split("::", 1)
        raw_path = raw_path.strip()
        goal = goal.strip()
        if not goal:
            return "Missing goal after '::'."
        try:
            workspace_root = resolve_workspace_root(raw_path)
        except WorkspaceError as exc:
            return f"Invalid workspace path: {exc}"
        if not workspace_root.is_dir():
            return f"Workspace '{workspace_root}' does not exist or is not a folder."

        self._start_session(workspace_root, goal)
        return _WORKING_INDICATOR

    async def _handle_natural_language_start(self, text: str) -> Tuple[bool, str]:
        """Resolve the owner's currently-selected 'Thu muc nguon' through
        the public Host API (Contract muc 11: knowledge_source_manager.
        get_runtime_context -> active_source_id, then the already-allowed
        list_folders) and auto-start a WorkSession there, using `text`
        verbatim as the goal. Never guesses a source/folder that wasn't
        the one the owner actually selected in the GUI -- any ambiguity
        or missing selection returns a consumed=True guidance message
        instead of picking on the owner's behalf."""
        try:
            ctx_resp = await self._invoke("knowledge_source_manager", "get_runtime_context", {})
        except Exception:  # noqa: BLE001 -- host_call failure must not crash the turn
            return True, _NEEDS_WORKSPACE_MESSAGE
        if not getattr(ctx_resp, "success", False):
            return True, _NEEDS_WORKSPACE_MESSAGE

        active_source_id = ((ctx_resp.data or {}).get("runtime_context") or {}).get("active_source_id")
        if not active_source_id:
            return True, _NEEDS_WORKSPACE_MESSAGE

        try:
            folders_resp = await self._invoke("knowledge_source_manager", "list_folders", {"source_id": active_source_id})
        except Exception:  # noqa: BLE001
            return True, _NO_USABLE_FOLDER_MESSAGE
        if not getattr(folders_resp, "success", False):
            return True, _NO_USABLE_FOLDER_MESSAGE

        folders = (folders_resp.data or {}).get("folders") or []
        usable_paths = [f["path"] for f in folders if f.get("path")]
        if not usable_paths:
            return True, _NO_USABLE_FOLDER_MESSAGE
        if len(usable_paths) > 1:
            return True, _multiple_folders_message(text, usable_paths)

        try:
            workspace_root = resolve_workspace_root(usable_paths[0])
        except WorkspaceError as exc:
            return True, f"The current Source Folder is invalid: {exc}"
        if not workspace_root.is_dir():
            return True, f"The current Source Folder '{workspace_root}' does not exist or is not a folder."

        self._start_session(workspace_root, text)
        return True, _WORKING_INDICATOR

"""Persisted WorkSession state -- the on-disk source of truth for exactly
one active work session per module process (Contract muc 9.2 already
limits the Host to at most 1 active module process; this file additionally
limits that 1 process to 1 WorkSession at a time, per V1 scope).

All paths are derived from the `module_data_root` injected into `boot()`
by the Host -- never CWD, never a hard-coded A Meo path (Contract muc 2.3).
Every write is atomic (`os.replace` after a temp-file write) so a process
kill mid-write can never leave a half-written session file behind.

Agent-first rewrite (owner directive): the session no longer tracks
pivot/affirmative-reply/goal-requires-mutation bookkeeping computed by
Python heuristics. It tracks the real `conversation` transcript (what the
model actually reasons over every turn) and a content-agnostic dangerous-
action signature used only for a real side-effect safety gate -- never as
an intent guess.

Workbench V2 (owner directive, "kim tay khong kim dau"): the no-progress/
oscillation repeat-tracking fields (`last_action_signature`,
`repeated_action_count`, `action_signature_history`) are REMOVED -- that
mechanism was Python judging "does the model's strategy look stuck" and
interrupting it, which is cognitive substitution for something a capable
model already does on its own by reading its own observation.
`safety.MAX_STEPS` is now the session's only step-budget concept; this file
carries no other notion of "cognitive progress" at all."""
from __future__ import annotations

import json
import os
import time
import uuid
from dataclasses import asdict, dataclass, field
from pathlib import Path
from typing import Any, Dict, List, Optional

STATUS_IDLE = "IDLE"
STATUS_RUNNING = "RUNNING"
STATUS_WAITING = "WAITING"
STATUS_COMPLETE = "COMPLETE"
STATUS_FAILED = "FAILED"
STATUS_CANCELLED = "CANCELLED"

TERMINAL_STATUSES = {STATUS_COMPLETE, STATUS_FAILED, STATUS_CANCELLED}


@dataclass
class WorkSession:
    session_id: str
    workspace_root: str
    goal: str
    status: str = STATUS_IDLE
    # The real dialogue the model reasons over every turn -- {"role":
    # "owner"|"assistant", "text": str}. `goal` is also the first "owner"
    # entry. Replaces `owner_input`/`pending_question`/
    # `last_approved_question`: there is no separate bookkeeping of "the"

class SessionStore:
    """Atomic JSON persistence under the injected module_data_root."""

    def __init__(self, module_data_root: Path):
        self._root = Path(module_data_root)
        self._sessions_dir = self._root / "sessions"
        self._sessions_dir.mkdir(parents=True, exist_ok=True)
        self._pointer_file = self._root / "current_session.txt"

    @property
    def root(self) -> Path:
        """The injected module_data_root -- used by agent_loop to place the
        bounded diagnostic turn-trace log (trace.py) next to session data,
        never a CWD/__file__ guess."""
        return self._root

    def _session_path(self, session_id: str) -> Path:
        return self._sessions_dir / f"{session_id}.json"

    @staticmethod
    def _atomic_write(path: Path, text: str) -> None:
        """Real, confirmed failure mode (owner directive 2026-08-21, traced
        from actual logs across THIS module, knowledge_source_manager, and
        estimation on this same OneDrive-synced deployment): `os.replace`
        can raise `PermissionError` (Windows WinError 5 "Access is denied")
        when the destination is transiently locked by OneDrive/AV right
        after being written -- normally clears within milliseconds. A
        short, bounded retry absorbs that instead of aborting the whole
        save (and, upstream, leaving a WorkSession created-but-never-
        actually-started -- see `SessionStore.new_session`/capability.py's
        `_start_session`)."""
        tmp_path = path.with_name(path.name + f".tmp-{uuid.uuid4().hex}")
        tmp_path.write_text(text, encoding="utf-8")
        last_exc: Optional[PermissionError] = None
        for _ in range(5):
            try:
                os.replace(tmp_path, path)  # atomic on the same volume, POSIX and Windows alike
                return
            except PermissionError as exc:
                last_exc = exc
                time.sleep(0.05)
        tmp_path.unlink(missing_ok=True)
        raise last_exc

    def save(self, session: WorkSession) -> None:
        session.updated_at = time.time()
        payload = json.dumps(session.to_dict(), ensure_ascii=False, indent=2)
        self._atomic_write(self._session_path(session.session_id), payload)
        self._atomic_write(self._pointer_file, session.session_id)

    def load(self, session_id: str) -> Optional[WorkSession]:
        path = self._session_path(session_id)
        if not path.is_file():
            return None
        # Same transient-lock class documented on `_atomic_write` above (real,
        # confirmed on this OneDrive-synced deployment): a read racing a
        # concurrent os.replace() can hit a brief PermissionError on Windows.
        # `_atomic_write` already retries for writers; without a matching
        # retry here, that SAME race on the READ side would make a real,
        # still-live session look like "no session" to a caller (e.g.
        # capability.py's WAITING/RUNNING branches), which could
        # incorrectly start a brand-new session instead of resuming the
        # real one. A short bounded retry absorbs it exactly like the write
        # side already does, instead of only guessing at "corrupt JSON".
        for _ in range(5):
            try:
                data = json.loads(path.read_text(encoding="utf-8"))
                return WorkSession.from_dict(data)
            except PermissionError:
                time.sleep(0.05)
            except (OSError, json.JSONDecodeError):
                return None
        return None

    def get_current(self) -> Optional[WorkSession]:
        if not self._pointer_file.is_file():
            return None
        session_id = self._pointer_file.read_text(encoding="utf-8").strip()
        if not session_id:
            return None
        return self.load(session_id)

    def new_session(
        self,
        workspace_root: str,
        goal: str,
        *,
        status: str = STATUS_IDLE,
        protected_paths: Optional[List[str]] = None,
    ) -> WorkSession:
        """Creates AND persists a session in exactly ONE `save()` call.
        Deliberately accepts the desired initial `status` up front rather
        than making the caller create-then-immediately-mutate-then-save-
        again: two back-to-back saves on session creation was a real,
        confirmed bug (owner directive 2026-08-21) -- a transient
        `PermissionError` (see `_atomic_write`) on the SECOND save left a
        session parked forever at status=IDLE (persisted by the first
        save, which DID succeed) with its background task never spawned,
        since the caller only called `_spawn_task_if_needed` after that
        second save. One save means the create is now all-or-nothing:
        either the session exists already-RUNNING and gets its task
        spawned, or it never got persisted at all and the caller's
        `new_session()` call itself raises -- no more silently-orphaned
        IDLE session in between.

        The goal text becomes the first `conversation` entry (role=owner)
        -- there is no separate "goal vs conversation" bookkeeping; the
        goal IS just how this conversation started."""
        session = WorkSession(
            session_id=uuid.uuid4().hex[:12],
            workspace_root=workspace_root,
            goal=goal,
            status=status,
            conversation=[{"role": "owner", "text": goal}],
            protected_paths=list(protected_paths or []),
        )
        self.save(session)
        return session

"""Workspace root resolution & sandbox boundary enforcement -- the ONLY
place path-escape decisions get made for this module. Every file_tools /
command_runner call goes through `resolve_inside` before touching disk;
nothing else in this package is allowed to build a filesystem path on its
own. Pure `pathlib`, no CWD dependency, no hard-coded OS path separators
(Public Module Contract v1, muc 2.3/2.4)."""
from __future__ import annotations

from pathlib import Path


class WorkspaceError(Exception):
    """Raised for anything that would resolve outside the workspace root
    (traversal, absolute-path injection, or a symlink pointing out) -- this
    is a hard reject, never something an owner can approve around, because
    it is a contract boundary rather than a risk judgement."""


def resolve_workspace_root(raw_path: str) -> Path:
    """Turn user-supplied text into an absolute, symlink-resolved
    directory path. Caller decides whether it must already exist."""
    if not raw_path or not raw_path.strip():
        raise WorkspaceError("Workspace path trong.")
    return Path(raw_path.strip()).expanduser().resolve()


def resolve_inside(workspace_root: Path, relative_path: str) -> Path:
    """Resolve `relative_path` (as supplied in an ACT step's params)
    against `workspace_root` and guarantee the final, symlink-resolved
    path is still inside it. `Path.resolve()` follows symlinks, so a
    symlink inside the workspace that points outside is caught by the same
    `relative_to` check as a plain `../..` traversal or an injected
    absolute path (joining an absolute path onto workspace_root simply
    replaces it, per `pathlib` semantics, so the check still fires)."""
    if relative_path is None:
        raise WorkspaceError("Thieu path.")
    candidate = (workspace_root / relative_path).resolve()
    try:
        candidate.relative_to(workspace_root)
    except ValueError as exc:
        raise WorkspaceError(f"Path '{relative_path}' nam ngoai workspace root.") from exc
    return candidate

class UnsupportedFormatError(Exception):

def _looks_binary(path: Path, sniff_bytes: int = 8000) -> bool:
    try:
        chunk = path.read_bytes()[:sniff_bytes]
    except OSError:
        return False
    if b"\x00" in chunk:
        return True
    try:
        chunk.decode("utf-8")
    except UnicodeDecodeError:
        return True
    return False


def _is_unsupported(path: Path) -> bool:
    if path.suffix.lower() in _KNOWN_UNSUPPORTED_EXTENSIONS:
        return True
    return _looks_binary(path)


def _format_mtime(ts: float) -> str:
    try:
        return datetime.datetime.fromtimestamp(ts).strftime("%Y-%m-%d %H:%M")
    except (OSError, OverflowError, ValueError):
        return "unknown"


# Workbench V2 (owner directive, search-tool audit -- live evidence: a real
# manuscript workspace had content up to chapter 37, but the agent only
# ever saw evidence up to ~chapter 25-26): `list_tree`/`search_text` used to
# return bare relative paths with no real evidence of which candidate was
# actually the newest -- the model had no way to determine "latest" except
# by guessing from filenames, and a plain-order match cap could silently
# drop later/relevant candidates before the model ever saw them. Both tools
# now surface real filesystem modification time alongside every entry
# (genuine evidence, not a guess), and `search_text` sorts matches
# newest-first so "find the latest occurrence of X" is answerable directly
# from the result list without hardcoding any domain-specific "latest"
# concept.
_SEARCH_SCAN_MAX_FILES = 5000


def list_tree(workspace_root: Path, relative_path: str = ".", max_depth: int = 3) -> List[str]:
    root = resolve_inside(workspace_root, relative_path)
    entries: List[str] = []
    base_depth = len(root.parts)
    for path in sorted(root.rglob("*")):
        if len(path.parts) - base_depth > max_depth:
            continue
        rel = str(path.relative_to(workspace_root))
        if path.is_dir():
            entries.append(rel + "/")
        else:
            try:
                mtime = path.stat().st_mtime
            except OSError:
                mtime = None
            entries.append(rel if mtime is None else f"{rel} (modified: {_format_mtime(mtime)})")
        if len(entries) >= MAX_LIST_ENTRIES:
            entries.append("... (truncated, con nhieu hon -- thu hep path de xem tiep)")
            break
    return entries


def _extract_full_text(target: Path, relative_path: str) -> str:
    """The ONE place a supported file's COMPLETE text is materialized --
    shared by `search_text` (so it can actually find a hit anywhere in the
    real document, not just a truncated prefix) and `read_file` (so its
    offset/length pagination below operates on the real, complete
    document). Raises exactly like the old per-caller checks did."""
    if target.suffix.lower() == ".docx":
        return docx_tools.read_docx_text(target)
    if target.suffix.lower() == ".xlsx":
        return xlsx_tools.read_xlsx_text(target)
    if _is_unsupported(target):
        raise UnsupportedFormatError(relative_path)
    data = target.read_bytes()[:MAX_READ_BYTES]
    return data.decode("utf-8", errors="replace")


def _find_all_offsets(haystack_lower: str, needle_lower: str, max_count: int, start: int = 0) -> List[int]:
    """Every occurrence of `needle_lower` in `haystack_lower` starting from
    character position `start`, in order, up to `max_count` -- returns MORE
    than max_count found (caller reports that honestly) only in the sense
    that this stops scanning once it has max_count + 1 (so callers can tell
    "exactly max_count" from "more exist" without an unbounded scan of a
    huge document). `start` lets a caller resume scanning a SINGLE file
    past whatever it already saw in an earlier call (see `search_text`'s
    `start_offset` param) instead of only ever seeing the first
    `max_count` occurrences from position 0 forever."""
    offsets: List[int] = []
    pos = max(0, start)
    step = max(1, len(needle_lower))
    while len(offsets) <= max_count:
        idx = haystack_lower.find(needle_lower, pos)
        if idx == -1:
            break
        offsets.append(idx)
        pos = idx + step
    return offsets


def search_text(
    workspace_root: Path, query: str, relative_path: str = ".", start_offset: int = 0,
) -> Dict[str, Any]:
    """Returns {"matches": [...], "skipped_unsupported": [...],
    "truncated": bool}. Each match is
    {"path", "modified", "match_count", "occurrences": [{"offset",
    "snippet"}, ...], "more_occurrences": bool, "next_occurrence_offset":
    int|None} -- multi-hit AND multi-source by construction (every matching
    file in the scope is returned, not just the first one; every occurrence
    up to MAX_OCCURRENCES_PER_FILE is returned per file, with its real
    character OFFSET into the file's extracted text so a follow-up
    `read_file(path, offset=...)` lands exactly there). Files this module
    cannot actually inspect as text are listed in `skipped_unsupported`
    rather than silently contributing to a false "no matches anywhere"
    result. Matches are sorted NEWEST-FIRST by real filesystem modification
    time, so "which of these is the latest" is answerable from real
    evidence, not a guess.

    `start_offset` (default 0, same character-offset unit `read_file`
    already uses): resumes occurrence-scanning of EVERY file in scope from
    that character position instead of position 0 -- this is what lets a
    caller reach occurrences PAST the per-file `MAX_OCCURRENCES_PER_FILE`
    cap in a long document (a file's own `next_occurrence_offset`, present
    whenever `more_occurrences` is true, is the value to pass back in as
    `start_offset` on the next call, most precisely when `relative_path`
    is narrowed to that one file). Omitting it (or leaving it 0) is
    unchanged prior behavior -- a fresh scan from the start of every file,
    exactly as before this parameter existed."""
    root = resolve_inside(workspace_root, relative_path)
    start_offset = max(0, int(start_offset or 0))
    hits: List[Tuple[float, Dict[str, Any]]] = []
    skipped_unsupported: List[str] = []
    if not query:
        return {"matches": [], "skipped_unsupported": [], "truncated": False}
    query_lower = query.lower()
    candidates = [root] if root.is_file() else sorted(root.rglob("*"))
    scanned = 0
    scan_truncated = False
    for path in candidates:
        if not path.is_file():
            continue
        scanned += 1
        if scanned > _SEARCH_SCAN_MAX_FILES:
            scan_truncated = True
            break
        rel = str(path.relative_to(workspace_root))
        name_hit = query_lower in path.name.lower()
        text = None
        if path.suffix.lower() in (".docx", ".xlsx"):
            try:
                text = _extract_full_text(path, rel)
            except Exception:  # noqa: BLE001 -- unreadable/corrupt -> can't inspect, not "no match"
                if not name_hit:
                    skipped_unsupported.append(rel)
                    continue
        elif _is_unsupported(path):
            if not name_hit:
                skipped_unsupported.append(rel)
                continue
        else:
            try:
                text = path.read_text(encoding="utf-8", errors="ignore")
            except OSError:
                text = None

        offsets: List[int] = []
        total_occurrences = 0
        if text is not None:
            offsets_found = _find_all_offsets(
                text.lower(), query_lower, MAX_OCCURRENCES_PER_FILE, start=start_offset,
            )
            total_occurrences = len(offsets_found)
            offsets = offsets_found[:MAX_OCCURRENCES_PER_FILE]

        # A file whose only occurrences are all before `start_offset` (the
        # caller already paged past them) correctly drops out of THIS
        # call's results -- name_hit alone (no start_offset relevance) can
        # still surface it.
        if not (name_hit or offsets):
            continue

        try:
            mtime = path.stat().st_mtime
        except OSError:
            mtime = 0.0

        occurrences = []
        if text is not None:
            for off in offsets:
                start = max(0, off - SNIPPET_RADIUS_CHARS)
                end = min(len(text), off + len(query) + SNIPPET_RADIUS_CHARS)
                snippet = text[start:end].replace("\n", " ").strip()
                occurrences.append({"offset": off, "snippet": snippet})

        has_more = total_occurrences > MAX_OCCURRENCES_PER_FILE
        hits.append((mtime, {
            "path": rel,
            "modified": _format_mtime(mtime),
            "match_count": total_occurrences,
            "occurrences": occurrences,
            "more_occurrences": has_more,
            # Character offset to pass back as `start_offset` (with
            # `relative_path` narrowed to this exact file) to continue
            # scanning THIS file past the occurrences already shown here --
            # same "resume point" shape as read_file's `next_offset`. None
            # when there is nothing left to page to.
            "next_occurrence_offset": (offsets[-1] + max(1, len(query))) if has_more else None,
        }))

    hits.sort(key=lambda h: h[0], reverse=True)
    match_truncated = len(hits) > MAX_SEARCH_MATCHES
    matches = [entry for _, entry in hits[:MAX_SEARCH_MATCHES]]
    return {
        "matches": matches,
        "skipped_unsupported": skipped_unsupported,
        "truncated": scan_truncated or match_truncated,
    }


# Default pagination window for `read_file` when the caller doesn't specify
# `length` -- generous enough for a normal "read around this offset" call,
# but explicit and honest (via `has_more`/`next_offset`) rather than a
# blind full-document dump, so a multi-million-character document is still
# navigable one real, bounded chunk at a time.
DEFAULT_READ_LENGTH = 20_000


def read_file(
    workspace_root: Path, relative_path: str, offset: int = 0, length: Optional[int] = None,
) -> Dict[str, Any]:
    """Returns {"content", "total_chars", "returned_start", "returned_end",
    "returned_chars", "has_more", "next_offset"} -- never a bare string, so
    a caller can never mistake a partial read for "the whole file" (no
    silent truncation). `offset`/`length` are character offsets into the
    file's fully-extracted text (same units `search_text`'s occurrence
    `offset` already uses, so a hit can be read directly via
    `read_file(path, offset=hit_offset)`). A negative `offset` counts from
    the end (Python slice convention), e.g. `offset=-2000` reads the last
    ~2000 chars without the caller needing to know `total_chars` first."""
    target = resolve_inside(workspace_root, relative_path)
    if not target.is_file():
        raise FileNotFoundError(f"'{relative_path}' khong phai file.")
    full_text = _extract_full_text(target, relative_path)
    total_chars = len(full_text)

    if offset < 0:
        offset = max(0, total_chars + offset)
    offset = min(offset, total_chars)
    window = DEFAULT_READ_LENGTH if length is None else max(0, int(length))
    end = min(offset + window, total_chars)

    return {
        "content": full_text[offset:end],
        "total_chars": total_chars,
        "returned_start": offset,
        "returned_end": end,
        "returned_chars": end - offset,
        "has_more": end < total_chars,
        "next_offset": end if end < total_chars else None,
    }


_NO_OVERWRITE_FORMATS = {".docx", ".xlsx"}


def write_file(
    workspace_root: Path, relative_path: str, content: str,
    protected_paths: Optional[Set[str]] = None,
) -> None:
    target = resolve_inside(workspace_root, relative_path)
    if target.suffix.lower() in _NO_OVERWRITE_FORMATS:
        raise UnsupportedFormatError(
            relative_path,
            f"'{relative_path}' la file {target.suffix} -- write_file (ghi de toan bo "
            f"bang plain text) se pha hong file nay. Dung patch_file de thay the noi "
            f"dung ben trong, giu nguyen cau truc file.",
        )
    # Only a pre-existing file is ever protected -- writing a brand-new
    # path (Workbench's own fresh output) is always allowed regardless.
    if target.exists():

"""DOCX read/search/patch support for Workbench's file tools.

Uses `python-docx` from the SAME interpreter the module process runs under
(Contract muc 10: module child processes default to `sys.executable`,

import os
import uuid
from pathlib import Path
from typing import Iterator, List, Tuple

try:
    from docx import Document
    _DOCX_AVAILABLE = True
except ImportError:  # pragma: no cover -- exercised only when python-docx is absent
    Document = None  # type: ignore[assignment]
    _DOCX_AVAILABLE = False

# Workbench V2 (owner directive, long-document audit): raised from
# 200_000 -- that was a SILENT truncation on the extraction side itself,
# meaning a manuscript past 200k chars could never be fully searched or
# read no matter how `read_file` was called. This is now a last-resort
# safety ceiling (a genuinely pathological file), not a normal operating
# limit -- real navigation is `file_tools.read_file`'s explicit
# offset/length pagination, applied AFTER this full extraction.
MAX_TEXT_CHARS = 5_000_000


class DocxUnavailableError(RuntimeError):
    """python-docx is not installed in this module process's interpreter."""


def is_docx_available() -> bool:
    return _DOCX_AVAILABLE


def _require_docx() -> None:
    if not _DOCX_AVAILABLE:
        raise DocxUnavailableError(
            "python-docx khong co san trong moi truong chay Workbench -- "
            "khong the doc/tim/sua file .docx."

def _iter_table_paragraphs(tables, label: str) -> Iterator[Tuple[str, object]]:
    for table in tables:
        for row in table.rows:
            for cell in row.cells:
                for p in cell.paragraphs:
                    yield (label, p)
                yield from _iter_table_paragraphs(cell.tables, label)  # nested tables


def _iter_all_paragraphs(document) -> Iterator[Tuple[str, object]]:
    """Every paragraph the document actually renders text in: body, table
    cells (incl. nested tables), headers, footers -- the 4 required
    surfaces, and nothing else (no comments/footnotes in V1 scope)."""
    for p in document.paragraphs:
        yield ("body", p)
    yield from _iter_table_paragraphs(document.tables, "table")
    for section in document.sections:
        for part, label in ((section.header, "header"), (section.footer, "footer")):
            if part is None:
                continue
            for p in part.paragraphs:
                yield (label, p)
            yield from _iter_table_paragraphs(part.tables, f"{label}_table")


def _open_document(path: Path):
    """A corrupt/not-actually-docx file must become a clean, caught
    ValueError (agent_loop.py's ACT except tuple), never a raw
    `zipfile.BadZipFile`/python-docx-internal exception escaping uncaught
    out of the background task (owner directive 2026-08-21, aligned with
    the same hardening in xlsx_tools.py)."""
    try:
        return Document(str(path))
    except Exception as exc:  # noqa: BLE001 -- deliberately broad: "can't open this as docx"
        raise ValueError(f"Khong the mo '{path.name}' nhu file .docx hop le: {exc}") from exc


def read_docx_text(path: Path) -> str:
    """Human-readable extracted text, NOT raw ZIP bytes -- one line per
    paragraph across body/table/header/footer, in document order."""
    _require_docx()
    document = _open_document(path)
    lines = [p.text for _, p in _iter_all_paragraphs(document) if p.text]
    text = "\n".join(lines)
    return text[:MAX_TEXT_CHARS]


def _run_boundaries(runs) -> List[Tuple[int, int]]:
    """Cumulative [start, end) character offsets of each run's CURRENT
    text within the paragraph's concatenated text."""
    bounds = []
    pos = 0
    for r in runs:
        length = len(r.text)
        bounds.append((pos, pos + length))
        pos += length
    return bounds


def _allocate_replacement(new_text: str, shares: List[int]) -> List[str]:
    """Splits `new_text` into `len(shares)` chunks whose lengths are
    proportional to `shares` (how many original matched characters each
    participating run contributed), summing to EXACTLY len(new_text).
    Deterministic largest-remainder rounding, not a random/arbitrary
    split -- ties broken by run order, so the same input always produces
    the same allocation."""
    n = len(shares)
    total_old = sum(shares)
    if total_old <= 0 or not new_text:
        base = [0] * n
        if new_text and n:
            base[0] = len(new_text)  # degenerate (old_text was empty-ish); keep it simple/deterministic
    else:
        raw = [len(new_text) * s / total_old for s in shares]
        base = [int(x) for x in raw]
        remainder = len(new_text) - sum(base)
        order = sorted(range(n), key=lambda i: (raw[i] - base[i]), reverse=True)
        for i in order[:remainder]:
            base[i] += 1
    chunks = []
    pos = 0
    for length in base:
        chunks.append(new_text[pos:pos + length])
        pos += length
    return chunks


def _apply_run_aware_replacement(runs, start: int, end: int, new_text: str) -> None:
    """Replaces the [start, end) character span (already located in the
    paragraph's concatenated text) with `new_text`, touching ONLY the run
    objects that actually overlap that span, and ONLY their `.text` --
    never their formatting (`.bold`/`.italic`/`.underline`/`.font.*`).
    A run entirely inside the span gets its full allocated chunk; a run
    that only partially overlaps keeps its own unmatched prefix/suffix
    text around that chunk. This is what makes a single-run match come
    out with that run's formatting untouched, and a multi-run match
    redistribute across exactly the runs it originally spanned instead of
    collapsing into the first one."""
    bounds = _run_boundaries(runs)
    participating = [i for i, (s, e) in enumerate(bounds) if e > start and s < end]
    shares = [min(bounds[i][1], end) - max(bounds[i][0], start) for i in participating]
    chunks = _allocate_replacement(new_text, shares)

    for pos_in_list, run_idx in enumerate(participating):
        run = runs[run_idx]
        run_start, run_end = bounds[run_idx]
        text = run.text
        prefix = text[: start - run_start] if run_start < start else ""
        suffix = text[end - run_start:] if run_end > end else ""
        run.text = prefix + chunks[pos_in_list] + suffix  # only .text changes -- rPr/formatting untouched


def _replace_in_paragraph(paragraph, old_text: str, new_text: str) -> int:
    """Replaces ALL occurrences of `old_text` in this paragraph, one at a
    time (re-scanning after each so shifted run boundaries stay correct),
    preserving formatting via `_apply_run_aware_replacement`."""
    count = 0
    while True:
        runs = paragraph.runs
        if not runs:
            break
        full_text = "".join(r.text for r in runs)
        idx = full_text.find(old_text)
        if idx == -1:
            break
        _apply_run_aware_replacement(runs, idx, idx + len(old_text), new_text)
        count += 1
    return count


def replace_text_in_docx(path: Path, old_text: str, new_text: str) -> int:
    """Replaces every occurrence of `old_text` with `new_text` across
    body/table/header/footer paragraphs. Returns the total replacement
    count; raises ValueError if `old_text` was not found anywhere (0
    replacements -- callers must never treat that as a silent no-op
    success). Saves atomically: to a temp file in the same directory
    first, `os.replace` onto the original only after a successful
    `Document.save()` -- the original is left untouched if save fails."""
    _require_docx()
    if not old_text:
        raise ValueError("old_text trong -- khong the patch file .docx.")
    document = _open_document(path)
    total = 0
    for _, paragraph in _iter_all_paragraphs(document):
        total += _replace_in_paragraph(paragraph, old_text, new_text)
    if total == 0:
        raise ValueError(f"Khong tim thay '{old_text}' trong file .docx nay.")

    tmp_path = path.with_name(path.name + f".tmp-{uuid.uuid4().hex}")
    try:
        document.save(str(tmp_path))
    except Exception:
        if tmp_path.exists():
            tmp_path.unlink(missing_ok=True)
        raise
    os.replace(tmp_path, path)
    return total

"""Subprocess execution for the agent loop's ACT step -- always
cwd=workspace root, always bounded by a timeout, output always captured
(never streamed to this child process's real stdout, which is reserved
for the Host protocol -- the A Meo Host runner redirects this process's
`sys.stdout` to stderr before any module code runs, so even a leaking
subprocess pipe here cannot corrupt the protocol)."""
from __future__ import annotations

import asyncio
import subprocess
from pathlib import Path
from typing import Any, Dict, List, Union

DEFAULT_TIMEOUT_SECONDS = 60
MAX_TIMEOUT_SECONDS = 180
MAX_OUTPUT_CHARS = 20_000


def _run_sync(workspace_root: Path, command: Union[str, List[str]], timeout: int) -> Dict[str, Any]:
    try:
        completed = subprocess.run(
            command,
            cwd=str(workspace_root),
            shell=isinstance(command, str),
            capture_output=True,
            text=True,
            timeout=timeout,
        )
        return {
            "stdout": completed.stdout[-MAX_OUTPUT_CHARS:],
            "stderr": completed.stderr[-MAX_OUTPUT_CHARS:],
            "exit_code": completed.returncode,
            "timed_out": False,
        }
    except subprocess.TimeoutExpired as exc:
        return {
            "stdout": (exc.stdout or "")[-MAX_OUTPUT_CHARS:],
            "stderr": (exc.stderr or "")[-MAX_OUTPUT_CHARS:],
            "exit_code": None,
            "timed_out": True,
        }
    except OSError as exc:
        return {"stdout": "", "stderr": str(exc), "exit_code": None, "timed_out": False}


async def run_command(
    workspace_root: Path,
    command: Union[str, List[str]],
    timeout: int = DEFAULT_TIMEOUT_SECONDS,
) -> Dict[str, Any]:
    """Runs synchronously in a worker thread (`asyncio.to_thread`) so the
    module's single event loop keeps servicing the protocol (host_call
    replies, future handle_turn/status polls) while a long command runs."""
    bounded_timeout = max(1, min(int(timeout or DEFAULT_TIMEOUT_SECONDS), MAX_TIMEOUT_SECONDS))
    return await asyncio.to_thread(_run_sync, workspace_root, command, bounded_timeout)

"""Bounded per-turn diagnostic trace for the Workbench agent loop.

A lesson carried over from Estimation's / SonHaiFactory's `trace.py` (the
discipline, not their code): when a `run_session` loop runs long -- a real
case on record is a single turn whose own large tool result took ~362s on
local Qwen -- nothing on disk says WHICH decision / tool call the time went
into. `session.steps` records the outcome of each step but no per-step
timing, so investigating "why was that turn slow" is guesswork. This writes
one short line per loop milestone to a size-capped rotating file next to the
module's data, and mirrors to stderr (captured by the Core process
manager).

Diagnostics ONLY -- it is not read back by the loop, never written into
`session.conversation`/`session.steps`, and a logging failure must never
break a turn. It matches Workbench's existing "observability of a physical
event that already happened, never cognitive orchestration" rule
(capability.py `_current_activity_text`): it logs a per-turn id, step
index, elapsed seconds, the chosen decision `type`, the tool NAME (never
its params/path/query), tool duration + result LENGTH, and the terminal
return reason -- no observation text, no reply/summary text, no owner
message, no file contents, no paths.
"""
from __future__ import annotations

import itertools
import logging
import logging.handlers
import time
from pathlib import Path
from typing import Optional

_LOG = logging.getLogger("workbench.turn_trace")
_CONFIGURED = False
# Fresh id per run_session invocation, process-local, monotonic. NOT
# derived from session_id/owner -- the trace carries no identifying data.
# Also doubles as loop order within the process.
_TURN_SEQ = itertools.count(1)


def _ensure(module_data_root: Optional[Path]) -> None:
    global _CONFIGURED
    if _CONFIGURED:
        return
    _CONFIGURED = True
    _LOG.setLevel(logging.INFO)
    _LOG.propagate = False
    fmt = logging.Formatter("%(asctime)s %(message)s")
    try:
        root = Path(module_data_root) if module_data_root else (Path(__file__).resolve().parent / "data")
        root.mkdir(parents=True, exist_ok=True)
        fh = logging.handlers.RotatingFileHandler(
            root / "workbench_turn_trace.log", maxBytes=256_000, backupCount=2, encoding="utf-8"
        )
        fh.setFormatter(fmt)
        _LOG.addHandler(fh)
    except Exception:  # noqa: BLE001 -- never let trace setup break a turn
        pass
    try:
        sh = logging.StreamHandler()  # -> stderr (captured by the Host)
        sh.setFormatter(logging.Formatter("[WB-trace] %(message)s"))
        _LOG.addHandler(sh)
    except Exception:  # noqa: BLE001
        pass


class TurnTrace:
    """One instance per `run_session` invocation. Cheap; safe to call from
    anywhere in the loop; every method swallows its own errors."""

    def __init__(self, module_data_root: Optional[Path] = None, resume_step_count: int = 0):
        _ensure(module_data_root)
        self._t0 = time.monotonic()
        self.turn_id = f"{next(_TURN_SEQ):05d}"
        self.step = 0
        self.emit(f"LOOP START resume_from_step={resume_step_count}")

    def elapsed(self) -> float:
        return time.monotonic() - self._t0

    def emit(self, msg: str) -> None:
        try:
            _LOG.info(f"[{self.turn_id} +{self.elapsed():7.1f}s st{self.step:02d}] {msg}")
        except Exception:  # noqa: BLE001
            pass

    def step_begin(self, n: int) -> None:
        self.step = n
        self.emit("decide-call BEGIN")

    def decision_done(self, decision_type: Optional[str], dt: float) -> None:
        self.emit(f"decide-call END dt={dt:.1f}s -> type={decision_type!r}")

    def tool(self, name: Optional[str], dt: float, result_len: int, ok: bool) -> None:
        self.emit(f"tool {name!r} dt={dt:.1f}s result_len={result_len} ok={ok}")

    def note(self, msg: str) -> None:
        self.emit(msg)

    def ret(self, why: str) -> None:
        self.emit(f"LOOP RETURN ({why}) total={self.elapsed():.1f}s steps={self.step}")

"""Background agent loop -- exactly ONE asyncio.Task per active
WorkSession, started by capability.py from a `handle_turn` call and then
running independently of any further `handle_turn`. The frozen runtime
rule (`handle_turn` MUST stay fast, per Contract muc 6.2's 60s timeout) is
what this file exists to satisfy: nothing in here is ever awaited by
`execute()`.

Agent-first rewrite (owner directive): the model owns understanding
intent, conversing, planning, and tool choice. Every turn it sees the
FULL conversation transcript (not a 3-step window) and returns exactly one
of 3 decision kinds -- `reply` (talk to the owner, pause), `tool_call`
(act), `complete` (claim the goal is done). Python's job per step is
dispatch + verification: run the real tool, track real mutation evidence,
and hold the model to its own `complete.mutated` claim against that
evidence -- never guess from the goal's wording what it "should" have
done.

Workbench V2 ("kim tay khong kim dau" -- owner directive): read-only
exploration (list_tree/search_text/read_file) runs freely, as many times
and in whatever order the model chooses, with NO Python guard pausing it
to ask the owner "still stuck?" -- a capable model reading its own real
observation is trusted to widen a search, try a different query, or
cross-check candidates on its own, exactly like it would in a normal
conversation. The only resource-limit safety nets left are: `MAX_STEPS`
(one simple hard ceiling against genuine runaway/token-explosion -- a
circuit breaker, never a strategy controller), `MAX_CONSECUTIVE_MALFORMED`
(machine-protocol integrity: the model isn't following the JSON contract),

dangerous-action confirmation round-trip (a real side-effect safety gate,
content-agnostic identity check, never a classifier of what the owner

from __future__ import annotations

import asyncio
import json
import re
import time

# Tools whose SUCCESS is unconditional mutation evidence (Contract-adjacent
# false-COMPLETE gate). run_command is deliberately excluded -- an exit
# code alone doesn't prove a file changed; see the filesystem fingerprint
# below for its real evidence path.
_UNCONDITIONAL_MUTATION_TOOLS = {"write_file", "patch_file", "delete_path", "make_dir"}

# Bounded filesystem fingerprint used ONLY to detect a real mutation caused
# by a successful run_command. Does NOT blindly trust exit_code==0 -- it
# additionally requires an actual before/after difference in
# (relpath -> mtime_ns, size) across the whole workspace. Same bounded-scan
# shape as capability.py's `_snapshot_existing_paths` (files only; capped
# so a huge workspace can't turn every run_command into a slow scan).
_FS_FINGERPRINT_MAX_ENTRIES = 5000

# How many recent tool-call steps are shown IN FULL in the observation --
# older ones are still accounted for via the cumulative progress summary
# (real counters, not a lossy re-summarization), just not replayed
# verbatim. Workbench V2: raised from 8 -- with the no-progress/
# oscillation guard removed and MAX_STEPS raised (free, wider exploration
# is now expected), an 8-step window risked the model losing visibility
# into its OWN earlier read-only exploration (which files it already
# checked) purely because that history scrolled out of view, not because
# it stopped mattering.
_RECENT_STEPS_WINDOW = 20

_RECENT_CONVERSATION_WINDOW = 20

_FULL_DETAIL_ASSISTANT_TURNS = 2

# Deterministic truncation only -- keeps enough of an aged-out assistant
# turn's own real text for continuity/gist (what was discussed, roughly
# what was concluded), but removes the bulk of it (including whatever
# specific numbers/quotes it contained) so it can no longer be copied
# verbatim as if it were fresh workspace evidence. Same shape/spirit as
# `_COMPACT_SNIPPET_CHARS` for search_text occurrences -- compression,
# not deletion.
_ASSISTANT_STUB_CHARS = 150


def _compact_assistant_text(text: str) -> str:
    if len(text) <= _ASSISTANT_STUB_CHARS:
        return text
    return (
        text[:_ASSISTANT_STUB_CHARS].rstrip()
        + "... [cau tra loi cu cua ban, DA RUT GON -- ngu canh hoi thoai, "
        "KHONG phai bang chung workspace]"
    )


_MALFORMED_ECHO_MAX_CHARS = 4000

_FULL_DETAIL_TOOL_CALLS = 2

# search_text's compact form keeps every occurrence's real OFFSET (an int,
# essentially free) plus a truncated snippet of this many chars -- short
# enough to keep compact meaningfully smaller than the full form
# (file_tools.SNIPPET_RADIUS_CHARS*2 per occurrence there), but never
# zero: "compact" means the evidence is COMPRESSED, not deleted down to a
# bare match_count (that gap is exactly what let a real production
# session fabricate chapter content once its real search_text evidence
# aged out of full-detail replay -- see test_search_text_pagination.py /
# test_evidence_compaction_and_repeat_notice.py).
_COMPACT_SNIPPET_CHARS = 60

_JSON_BLOCK_RE = re.compile(r"\{.*\}", re.DOTALL)


def _extract_json(text: str) -> Dict[str, Any]:
    match = _JSON_BLOCK_RE.search(text or "")
    if not match:
        raise safety.DecisionError("Khong tim thay JSON trong phan hoi AI.")
    # strict=False: accept raw control characters (literal newline/tab) inside
    # JSON string values -- confirmed real cause of a malformed-decision
    # repair round-trip (long multi-paragraph reply text with an unescaped
    # newline inside the "text" string, rejected by strict-mode json.loads).
    # Still rejects genuinely malformed JSON (missing delimiters/braces).
    return json.loads(match.group(0), strict=False)


def _fs_fingerprint(workspace_root: Path) -> Dict[str, Tuple[int, int]]:
    fingerprint: Dict[str, Tuple[int, int]] = {}
    try:
        for path in workspace_root.rglob("*"):
            if not path.is_file():
                continue
            try:
                rel = path.relative_to(workspace_root).as_posix()
                stat = path.stat()
                fingerprint[rel] = (stat.st_mtime_ns, stat.st_size)
            except OSError:
                continue
            if len(fingerprint) >= _FS_FINGERPRINT_MAX_ENTRIES:
                break
    except OSError:
        pass
    return fingerprint


def _build_progress_summary(session) -> Optional[str]:
    """Very short cumulative-progress line covering the WHOLE session, not
    just the windowed recent steps below -- reuses fields that are already
    tracked/persisted regardless of window size, so nothing is silently
    lost when older steps scroll out of the recent-steps window."""
    if session.successful_mutation_count <= 0 and not session.touched_paths:
        return None
    parts = [f"da thuc hien {session.successful_mutation_count} mutation thanh cong"]
    if session.total_replacements:
        parts.append(f"{session.total_replacements} vi tri da thay the")
    if session.touched_paths:
        shown = session.touched_paths[:5]
        remainder = len(session.touched_paths) - len(shown)
        paths_text = ", ".join(shown)
        if remainder > 0:
            paths_text += f" (+{remainder} file khac)"
        parts.append(f"file da xu ly: {paths_text}")
    return "Tien do da tich luy tu truoc den gio: " + "; ".join(parts) + "."


def _aggregate_search_evidence_chains(session) -> Dict[Tuple[str, str], Dict[str, Any]]:
    """Groups every step's `search_evidence` (see `_execute_sync_action`'s
    search_text branch) by (query, path) -- the 2 params a real
    pagination chain keeps constant across calls, only `start_offset`
    changes. Scans ALL of `session.steps`, not just the windowed recent
    ones -- each record is tiny (offsets + short snippets), so this stays
    cheap even for a long session, and is exactly why it survives step
    compaction/aging that the raw replayed text does not. Later steps in
    the SAME chain overwrite `terminal`/`next_offsets` (the most recent
    probe is the authoritative one for "where do we stand now"); matches
    from every step accumulate (dedup by (path, offset))."""
    chains: Dict[Tuple[str, str], Dict[str, Any]] = {}
    for step in session.steps:
        se = step.get("search_evidence")
        if not se:
            continue
        key = (se["query"], se["path"])
        chain = chains.setdefault(key, {"offsets": {}, "terminal": False, "next_offsets": {}})
        for m in se["matches"]:
            chain["offsets"][(m["path"], m["offset"])] = m["snippet"]
        chain["terminal"] = se["terminal"]
        chain["next_offsets"] = {
            fpath: finfo["next_occurrence_offset"]
            for fpath, finfo in se["files"].items()
            if finfo.get("more_occurrences")
        }
    return chains


def _build_search_evidence_chains(session) -> List[str]:
    """Renders 1 compact, ALWAYS-full block per search_text pagination
    chain -- never itself subject to `_FULL_DETAIL_TOOL_CALLS` aging,
    because it is derived fresh from every step's small `search_evidence`
    record rather than replaying any step's bulky text. Live defect fixed
    here (2026-08-25/26, evidence-traced: a 10-step chain with cursor
    0->42845->83707->115299->145000->120000->0->0 -- the tool itself had
    already reached and confirmed EOF at offset 145000, but once the
    earlier steps holding that evidence compacted, the model lost track,
    backtracked, and replied with an incomplete enumeration). This block
    is the fix: a small always-visible summary stating exactly which
    offsets are already known and whether the chain is terminal, so the
    model never has to reconstruct that from scattered/aged step text --
    it does not decide anything for the model, it only keeps the real
    facts from disappearing."""
    chains = _aggregate_search_evidence_chains(session)
    if not chains:
        return []
    lines = [
        "Chuoi tim kiem (search_text) da biet trong phien nay -- TONG HOP "
        "DAY DU, khong mat du cac step rieng le o tren da bi rut gon:"
    ]
    for (query, path), chain in chains.items():
        offsets_sorted = sorted(chain["offsets"].items(), key=lambda kv: (kv[0][0], kv[0][1]))
        if chain["terminal"]:
            state_text = (
                "DA TERMINAL (da xac nhan khong con vi tri nao khac -- "
                "KHONG can quay lai offset 0 hay lap lai tim kiem nay)"
            )
        elif chain["next_offsets"]:
            continuations = ", ".join(
                f"'{fpath}' tiep tuc tu start_offset={off}" for fpath, off in chain["next_offsets"].items()
            )
            state_text = f"CHUA terminal -- con continuation: {continuations}"
        else:
            state_text = "CHUA terminal"
        lines.append(f"- query='{query}' path='{path}': {len(offsets_sorted)} vi tri da biet -- {state_text}")
        for (fpath, off), snippet in offsets_sorted:
            lines.append(f"    [{fpath} @ offset {off}] {snippet}")
    return lines


def _build_observation(session) -> str:
    # `session.goal` is ONLY the opening line this session started with --
    # historical context, NOT the current objective (confirmed against
    # SessionStore.new_session: goal is set once, never amended; a later
    # owner turn on a different topic in the same WAITING/RUNNING session
    # never updates it). Shown here UNCONDITIONALLY, independent of the
    # conversation window below, purely so that history survives once the
    # goal's own original conversation entry has scrolled out of the raw
    # window -- labeled explicitly as history so the model never mistakes
    # it for what the owner wants RIGHT NOW. The actual current request is
    # whatever the OWNER's most recent turn in the conversation below says
    # -- Python never decides this, the model reads it directly.
    lines = [
        f"Workspace: {session.workspace_root}",
        "",
        f"Phien nay bat dau bang cau lenh (lich su, THAM KHAO, KHONG chac "
        f"la yeu cau hien tai neu owner da noi gi khac o duoi): {session.goal}",
        "",
    ]
    progress_summary = _build_progress_summary(session)
    if progress_summary:
        lines.append(progress_summary)
        lines.append("")

    search_evidence_lines = _build_search_evidence_chains(session)
    if search_evidence_lines:
        lines.extend(search_evidence_lines)
        lines.append("")

    recent_steps = session.steps[-_RECENT_STEPS_WINDOW:]
    older_count = len(session.steps) - len(recent_steps)
    if older_count > 0:
        lines.append(f"(+{older_count} tool_call cu hon, xem tom tat tien do o tren)")
    if recent_steps:
        lines.append("Cac tool_call gan nhat (bang chung DA thu thap TRUOC DAY -- kiem tra "
                      "xem co dung khop voi dieu owner dang hoi o luot GAN NHAT ben duoi "
                      "khong, dung mac dinh la du neu cau hoi hien tai can bang chung khac/"
                      "moi hon, vd hoi ve 'moi nhat'/'hien tai'/'toan bo' ma bang chung duoi "
                      "day chi la 1 phan/1 goc do cu):")
        # Only the _FULL_DETAIL_TOOL_CALLS most-recent tool_call steps get
        # their full result replayed -- older ones (still within the
        # window) get the deterministic compact form computed at execution
        # time (see `_execute_sync_action`), never a re-summary of the
        # already-stored full text. Non-tool_call steps (malformed/
        # dangerous-pending/complete-rejected/etc.) are already small and
        # always shown as-is.
        tool_call_positions = [
            i for i, s in enumerate(recent_steps) if s["decision"].get("type") == "tool_call"
        ]
        full_detail_positions = set(tool_call_positions[-_FULL_DETAIL_TOOL_CALLS:])
        for i, step in enumerate(recent_steps):
            is_tool_call = i in tool_call_positions
            show_full = (not is_tool_call) or (i in full_detail_positions)
            if show_full:
                # No truncation here -- `action_result` is exactly what the
                # tool itself already returned, each tool being responsible
                # for its own explicit, honest bound (see run_session).
                result_text = step["action_result"]
            else:
                result_text = step.get("action_result_compact") or step["action_result"]
            lines.append(
                f"- step {step['index']}: tool={step['decision'].get('tool')} "
                f"params={step['decision'].get('params')} ok={step['ok']} "
                f"result={result_text}"
            )
    else:
        lines.append("Chua co tool_call nao.")
    lines.append("")

    lines.append(
        "Cuoc hoi thoai voi owner (NGU CANH/YEU CAU, KHONG PHAI bang chung "
        "workspace -- ke ca luot [Ban (AI)] la chinh cau ban da tung noi, "
        "van chi la mot claim cu, khong tu dong la provenance cho fact "
        "workspace. Yeu cau HIEN TAI la luot [Owner] GAN NHAT ben duoi -- "
        "day la dieu ban PHAI xu ly trong luot nay, neu no khac chu de voi "
        "cau lenh mo dau o tren hay lam theo luot gan nhat, van co the tu "
        "lien he boi canh cu neu thay phu hop; neu khang dinh workspace can "
        "cho cau hoi nay khong duoc tool_call THUC SU nao o tren ho tro, "
        "dung tool_call de lay/xac nhan lai bang chung that truoc khi tra "
        "loi, KHONG chi dua lai claim cu):"
    )
    recent_conversation = session.conversation[-_RECENT_CONVERSATION_WINDOW:]
    older_conversation_count = len(session.conversation) - len(recent_conversation)
    if older_conversation_count > 0:
        lines.append(
            f"(+{older_conversation_count} luot trao doi cu hon voi owner, khong hien lai "
            "chi tiet -- can bang chung cu the thi tiep tuc "
            "dung search_text/read_file/list_tree.)"
        )
    # Only the _FULL_DETAIL_ASSISTANT_TURNS most-recent ASSISTANT turns
    # keep their full text; older ones get the deterministic stub above.
    # Owner turns are NEVER compacted, at any age, within this window.
    assistant_positions = [
        i for i, t in enumerate(recent_conversation) if t.get("role") != "owner"
    ]
    full_detail_assistant_positions = set(assistant_positions[-_FULL_DETAIL_ASSISTANT_TURNS:])
    for i, turn in enumerate(recent_conversation):
        is_owner = turn.get("role") == "owner"
        speaker = "Owner" if is_owner else "Ban (AI)"
        text = turn.get("text", "")
        if not is_owner and i not in full_detail_assistant_positions:
            text = _compact_assistant_text(text)
        lines.append(f"[{speaker}] {text}")

    return "\n".join(lines)


def _execute_sync_action(
    workspace_root: Path, decision: Dict[str, Any], protected_paths: Optional[set] = None,
) -> Tuple[str, Optional[int], Optional[str], Optional[Dict[str, Any]]]:
    """Returns (message, replaced_count, compact_message, search_evidence).
    `replaced_count` is only meaningful for `patch_file` (the false-
    COMPLETE gate's "N vi tri" evidence) -- every other tool returns None
    there. `compact_message` is a deterministic, structure-derived short
    form for tools whose full output can be large (`search_text`/
    `read_file`/`list_tree`) -- None for every other tool, meaning "no
    separate compact form, the full message is already short enough" (see
    `_build_observation`'s fallback). `search_evidence` (owner directive
    2026-08-26, evidence-chain fix) is a small structured record of this
    call's real matches -- ONLY `search_text` populates it, every other
    tool returns None -- persisted on the step so `_build_search_evidence_
    chains()` can reconstruct a full pagination chain's known offsets/
    terminal state EVEN ONCE the individual step ages out of full-detail
    replay (see that function's docstring for why this exists)."""
    tool = decision["tool"]
    params = decision.get("params") or {}

    if tool == "list_tree":
        entries = file_tools.list_tree(workspace_root, params.get("path", "."))
        full_message = "\n".join(entries)
        truncated = bool(entries) and entries[-1].startswith("... (truncated")
        counted = entries[:-1] if truncated else entries
        dir_count = sum(1 for e in counted if e.endswith("/"))
        file_count = len(counted) - dir_count
        compact_message = (
            f"[list_tree dang gon] {len(counted)} muc ({dir_count} thu muc, "
            f"{file_count} file)" + (", DA CAT BOT (con nhieu hon)" if truncated else ", day du")
            + " -- goi lai list_tree voi cung path de xem lai danh sach chi tiet."
        )
        return full_message, None, compact_message, None
    if tool == "search_text":
        result = file_tools.search_text(
            workspace_root, params.get("query", ""), params.get("path", "."),
            params.get("start_offset", 0) or 0,
        )
        if result["matches"]:
            lines = [f"Tim thay {len(result['matches'])} file co khop:"]
            for m in result["matches"]:
                header = f"- {m['path']} (modified: {m['modified']})"
                if m["match_count"]:
                    header += f" -- {m['match_count']} vi tri khop trong noi dung"
                    if m["more_occurrences"]:
                        header += (
                            f" (hien {len(m['occurrences'])} vi tri tinh tu start_offset hien tai, "
                            f"CON NUA -- goi lai search_text voi path='{m['path']}' va "
                            f"start_offset={m['next_occurrence_offset']} de xem tiep)"
                        )
                lines.append(header)
                for occ in m["occurrences"]:
                    lines.append(f"    [offset {occ['offset']}] ...{occ['snippet']}...")
        else:
            lines = ["Matches: (khong co)"]
        if result["skipped_unsupported"]:
            lines.append(
                f"Bo qua {len(result['skipped_unsupported'])} file khong doc duoc nhu van ban "
                f"(unsupported_format): " + ", ".join(result["skipped_unsupported"])
            )
        if result.get("truncated"):
            lines.append(
                "Luu y: con nhieu ket qua/file hon chua duoc hien thi het (da cat bot) -- "
                "neu can chac chan day du, thu hep 'path' xuong 1 thu muc cu the hon."
            )
        if result["matches"]:
            compact_lines = [f"Tim thay {len(result['matches'])} file co khop (dang gon, snippet rut gon):"]
            shown = result["matches"][:10]
            for m in shown:
                header = f"- {m['path']} (modified: {m['modified']})"
                if m["match_count"]:
                    header += f" -- {m['match_count']} vi tri khop"
                    if m["more_occurrences"]:
                        header += (
                            f" (CON NUA, goi lai voi start_offset="
                            f"{m['next_occurrence_offset']} de xem tiep)"
                        )
                compact_lines.append(header)
                for occ in m["occurrences"]:
                    short_snippet = occ["snippet"][:_COMPACT_SNIPPET_CHARS].strip()
                    compact_lines.append(f"    [offset {occ['offset']}] {short_snippet}...")
            remainder = len(result["matches"]) - len(shown)
            if remainder > 0:
                compact_lines.append(f"  (+{remainder} file khac co khop, chua liet ke o dang gon)")
            if result.get("truncated"):
                compact_lines.append("Luu y: con nhieu ket qua hon chua hien thi het.")
            compact_lines.append(
                "(Snippet o tren da rut gon -- goi lai search_text voi cung query/path/"
                "start_offset de xem lai doan van day du hon quanh cac offset nay.)"
            )
        else:
            compact_lines = lines
        # Structured evidence for THIS call, keyed by (query, path) -- the
        # exact 2 params that stay constant across a pagination chain
        # (only `start_offset` changes between calls). `matches` is a flat
        # list across every file this call touched, each with a small
        # snippet (same truncation as the compact form) -- small enough to
        # survive step-window/compaction cheaply. `files` carries each
        # touched file's own continuation state (`more_occurrences`/
        # `next_occurrence_offset`) so a later chain-summary can tell
        # "still has an unvisited continuation" from "confirmed no more
        # past this point" per file. `terminal` is TRUE only when THIS
        # call itself found nothing left to page through (either zero
        # matches at all -- the real "start_offset past EOF" signal -- or
        # every touched file's own `more_occurrences` is already False)
        # AND the file-count-level `truncated` flag isn't hiding more
        # files entirely.
        files_evidence: Dict[str, Dict[str, Any]] = {}
        matches_flat: List[Dict[str, Any]] = []
        for m in result["matches"]:
            files_evidence[m["path"]] = {
                "more_occurrences": m["more_occurrences"],
                "next_occurrence_offset": m.get("next_occurrence_offset"),
            }
            for occ in m["occurrences"]:
                matches_flat.append({
                    "path": m["path"], "offset": occ["offset"],
                    "snippet": occ["snippet"][:_COMPACT_SNIPPET_CHARS].strip(),
                })
        call_terminal = (
            not any(v["more_occurrences"] for v in files_evidence.values())
            and not result.get("truncated")
        )
        search_evidence = {
            "query": params.get("query", ""),
            "path": params.get("path", "."),
            "matches": matches_flat,
            "files": files_evidence,
            "terminal": call_terminal,
        }
        return "\n".join(lines), None, "\n".join(compact_lines), search_evidence
    if tool == "read_file":
        result = file_tools.read_file(
            workspace_root, params.get("path", ""), params.get("offset", 0) or 0, params.get("length"),
        )
        header = (
            f"[doc {result['returned_start']}-{result['returned_end']} / tong {result['total_chars']} chars]"
        )
        if result["has_more"]:
            header += f" -- CON NUA, doc tiep voi offset={result['next_offset']}"
        else:
            header += " -- DA DEN CUOI FILE"
        full_message = header + "\n" + result["content"]
        # Compact form: same navigation header (offset/total_chars/
        # has_more/next_offset -- exactly what's needed to re-read the same
        # range or continue past it) + a short content preview instead of
        # the full body. path/offset stay visible via the step's own
        # params, so re-calling read_file with the same args reproduces
        # this exactly.
        _COMPACT_PREVIEW_CHARS = 400
        preview = result["content"][:_COMPACT_PREVIEW_CHARS]
        compact_message = (
            header + " (dang gon, chi xem truoc "
            f"{min(_COMPACT_PREVIEW_CHARS, len(result['content']))} ky tu dau -- goi lai "
            "read_file voi cung path/offset de xem lai full noi dung)\n" + preview
        )
        return full_message, None, compact_message, None
    if tool == "write_file":
        file_tools.write_file(
            workspace_root, params.get("path", ""), params.get("content", ""),
            protected_paths=protected_paths,
        )
        return f"Da ghi '{params.get('path')}'.", None, None, None
    if tool == "patch_file":
        count = file_tools.patch_file(
            workspace_root, params.get("path", ""), params.get("old_text", ""), params.get("new_text", ""),
            protected_paths=protected_paths,
        )
        return f"Da patch '{params.get('path')}' ({count} cho thay the).", count, None, None
    if tool == "make_dir":
        file_tools.make_dir(workspace_root, params.get("path", ""))
        return f"Da tao thu muc '{params.get('path')}'.", None, None, None
    if tool == "delete_path":
        file_tools.delete_path(
            workspace_root, params.get("path", ""), bool(params.get("recursive")),
            protected_paths=protected_paths,
        )
        return f"Da xoa '{params.get('path')}'.", None, None, None
    raise safety.DecisionError(f"Tool '{tool}' khong co executor dong bo.")


async def _execute_action(
    workspace_root: Path, decision: Dict[str, Any], protected_paths: Optional[set] = None,
) -> Tuple[str, Optional[int], Optional[str], Optional[Dict[str, Any]]]:
    if decision["tool"] == "run_command":
        params = decision.get("params") or {}
        result = await command_runner.run_command(
            workspace_root,
            params.get("command", ""),
            params.get("timeout", command_runner.DEFAULT_TIMEOUT_SECONDS),
        )
        return json.dumps(result, ensure_ascii=False), None, None, None
    # Real confirmed bug: file_tools work (rglob directory scans, docx XML
    # parsing) used to run directly on the module's single event-loop
    # thread -- exactly like command_runner.run_command already avoids via
    # asyncio.to_thread, a slow/large ACT step here blocked that SAME loop
    # from servicing any concurrently-arriving handle_turn (incl. a plain
    # 'status' check), which could then hit the Host's 60s handle_turn
    # timeout and get the whole module process killed. Running it in a
    # worker thread keeps the loop free to answer 'status' (capability.py's
    # status path does no I/O of its own) in well under a second regardless
    # of how long the ACT step takes.
    return await asyncio.to_thread(_execute_sync_action, workspace_root, decision, protected_paths)


def _find_prior_identical_tool_call(session, signature: str) -> Optional[int]:
    """Returns the step `index` of the EARLIEST prior tool_call in this
    session whose own `safety.action_signature(tool, params)` matches
    `signature` exactly -- same tool, same params, byte-identical (a
    changed `start_offset`/`query`/`path` is a DIFFERENT signature, never
    flagged). Pure identity lookup over data already in `session.steps`,
    no new persisted state, no judgement about whether repeating was
    "useful" -- that stays entirely the model's call, this only surfaces
    the fact.

    Only SUCCESSFUL prior calls count (`step["ok"]`). The notice this
    feeds (`_repeat_notice`) asserts "re-running gives the exact same
    result, no new evidence" -- true for a repeated successful call (the
    real no-progress-loop case this exists for), but FALSE after a
    transient failure (a Windows/OneDrive sharing-lock on write_file/
    patch_file, a flaky run_command): retrying the identical call there is
    the correct move and genuinely can succeed. A prior FAILED identical
    step must not make a later successful retry look like a pointless
    repeat that produced nothing."""
    for step in session.steps:
        decision = step.get("decision") or {}
        if decision.get("type") != "tool_call":
            continue
        if not step.get("ok"):
            continue
        if safety.action_signature(decision.get("tool"), decision.get("params") or {}) == signature:
            return step.get("index")
    return None


def _repeat_notice(prior_index: int, continuation_hint: Optional[str] = None) -> str:
    """`continuation_hint` (owner directive 2026-08-26, evidence-chain
    fix): when the repeated call is a `search_text` with known pagination
    state, points the model at the chain's actual known continuation
    (or terminal state) INSTEAD of leaving it to guess -- pure fact
    reference from `search_evidence` already computed for this exact
    call, never a Python guess about domain/intent. None for every other
    tool/case, unchanged behavior."""
    base = (
        f"\n[LUU Y THUC TE: day la tool_call CO CUNG tool+params y het buoc "
        f"#{prior_index} truoc do trong phien nay -- ket qua thuc thi lai "
        f"cung se giong y het lan do, KHONG co bang chung moi tu lan goi "
        f"nay. Neu can bang chung KHAC, hay doi query/path/offset/tool "
        f"truoc khi goi lai. Neu ban dang co y goi lai de xem lai dung "
        f"bang chung cu do (vd bang chung cua buoc #{prior_index} da bi "
        f"rut gon truoc do va ban can xem lai chi tiet day du), ket qua "
        f"ben tren chinh la no.]"
    )
    if continuation_hint:
        base += f"\n[{continuation_hint}]"
    return base


def _search_repeat_continuation_hint(search_evidence: Optional[Dict[str, Any]]) -> Optional[str]:
    if not search_evidence:
        return None
    if search_evidence.get("terminal"):
        return (
            "Chuoi tim kiem nay (cung query/path) DA TERMINAL -- da xac nhan "
            "khong con vi tri nao khac, KHONG can lap lai hay quay ve offset 0."
        )
    next_offsets = {
        fpath: finfo["next_occurrence_offset"]
        for fpath, finfo in (search_evidence.get("files") or {}).items()
        if finfo.get("more_occurrences")
    }
    if not next_offsets:
        return None
    parts = ", ".join(f"'{fpath}' tiep tuc tu start_offset={off}" for fpath, off in next_offsets.items())
    return f"Continuation CHUA duyet cua chuoi tim kiem nay: {parts}."


def _append_assistant_message(session, text: str) -> None:
    session.conversation.append({"role": "assistant", "text": text})


# Deterministic tool -> owner-facing activity text, built ONLY from the
# tool name + its own params (real fields the model already sent in its
# `tool_call` decision) -- never a re-inference of what the model is
# "thinking". This is what capability.py's status path shows while
# RUNNING instead of the old static "..." placeholder, and what replaces
# the removed `progress` decision's narration without spending a second
# model call to produce it.
def describe_tool_activity(tool: str, params: Dict[str, Any]) -> str:
    params = params or {}
    path = params.get("path")
    if tool == "search_text":
        query = params.get("query")
        text = f"Dang tim '{query}'" if query else "Dang tim kiem"
        return text + (f" trong '{path}'" if path else "")
    if tool == "read_file":
        return f"Dang doc '{path}'" if path else "Dang doc file"
    if tool == "list_tree":
        return f"Dang liet ke '{path}'" if path else "Dang liet ke thu muc"
    if tool in ("write_file", "patch_file", "make_dir", "delete_path"):
        return f"Dang xu ly file '{path}'" if path else "Dang xu ly file"
    if tool == "run_command":
        command = params.get("command")
        return f"Dang thuc thi lenh: {command}" if command else "Dang thuc thi lenh"
    return "Dang xu ly"


async def run_session(session_id: str, store: SessionStore, invoke: InvokeFn) -> None:
    """The ONE background asyncio.Task per active WorkSession. Reloads the
    session from disk at the top of every iteration -- this is what lets a
    concurrent `handle_turn` (e.g. 'cancel') take effect between steps
    without any in-process locking, and is also exactly how a fresh
    process resumes a persisted RUNNING session after a restart: the first
    call into this function after boot just re-enters this same loop."""
    session = store.load(session_id)
    if session is None:
        return

    # Bounded diagnostic trace (trace.py) -- one per run_session entry
    # (which is also the resume entry point after a restart). Never read
    # back by the loop, never persisted into the session; a trace failure
    # is swallowed inside TurnTrace.
    tt = trace.TurnTrace(getattr(store, "root", None), resume_step_count=len(session.steps))

    while True:
        session = store.load(session_id) or session
        if session.status != STATUS_RUNNING:
            tt.ret(f"status={session.status}")
            return
        if len(session.steps) >= safety.MAX_STEPS:
            session.status = STATUS_FAILED
            session.result_summary = "Da vuot qua so buoc toi da cho phep (bounded agent steps)."
            store.save(session)
            tt.ret("max_steps")
            return

        tt.step_begin(len(session.steps) + 1)
        observation = _build_observation(session)

        _decide_t0 = time.monotonic()
        try:
            resp = await invoke("conversation", "chat", {
                "text": observation,
                "system_prompt": SYSTEM_PROMPT,
            })
        except Exception as exc:  # noqa: BLE001 -- an invoke() crash must not kill the process
            session.status = STATUS_FAILED
            session.result_summary = f"Loi goi AI: {exc}"
            store.save(session)
            tt.decision_done("invoke_raised", time.monotonic() - _decide_t0)
            tt.ret("invoke_exception")
            return

        if not getattr(resp, "success", False):
            session.status = STATUS_FAILED
            session.result_summary = f"AI invoke that bai: {getattr(resp, 'error', 'unknown')}"
            store.save(session)
            tt.decision_done("invoke_failed", time.monotonic() - _decide_t0)
            tt.ret("invoke_not_success")
            return

        ai_text = (resp.data or {}).get("response", "") if resp.data else ""

        try:
            decision = safety.validate_decision(_extract_json(ai_text))
        except (safety.DecisionError, json.JSONDecodeError) as exc:
            tt.decision_done("malformed", time.monotonic() - _decide_t0)
            session.consecutive_malformed += 1
            echoed = ai_text.strip()
            if len(echoed) > _MALFORMED_ECHO_MAX_CHARS:
                echoed = echoed[:_MALFORMED_ECHO_MAX_CHARS] + " ...(da cat bot)"
            session.steps.append({
                "index": len(session.steps) + 1,
                "decision": {"tool": "(malformed)"},
                "action_result": (
                    f"{exc} -- day la NGUYEN VAN ban vua tra loi (khong dung "
                    f"dinh dang JSON bat buoc o tren): {echoed!r}\nNeu noi dung "
                    f"nay DA la mot cau tra loi dung/du cho owner, ban co the "
                    f'DONG GOI LAI CHINH NOI DUNG DO vao dung 1 trong 4 dang '
                    f'JSON (thuong la {{"type": "reply", "text": "..."}}) thay '
                    f"vi viet lai tu dau. Nhung neu noi dung do CHUA du bang "
                    f"chung, lac de, hoac chua thuc su tra loi dung cau hoi cua "
                    f"owner, hay tiep tuc dung tool (search_text/read_file) "
                    f"hoac viet lai cho dung -- dinh dang JSON la bat buoc, "
                    f"noi dung van do ban tu quyet dinh."
                ),
                "verify": "malformed_decision",
                "ok": False,
            })
            if session.consecutive_malformed >= safety.MAX_CONSECUTIVE_MALFORMED:
                session.status = STATUS_FAILED
                session.result_summary = "AI lien tuc tra loi sai dinh dang JSON."
            store.save(session)
            if session.status == STATUS_FAILED:
                tt.ret("consecutive_malformed")
                return
            continue

        session.consecutive_malformed = 0
        tt.decision_done(decision["type"], time.monotonic() - _decide_t0)

        if decision["type"] == "reply":
            _append_assistant_message(session, decision["text"])
            session.status = STATUS_WAITING
            store.save(session)
            tt.ret("reply")
            return

        if decision["type"] == "complete":
            if decision["mutated"] and session.successful_mutation_count == 0:
                # False-COMPLETE gate: the model CLAIMS it mutated the
                # workspace, but zero mutations have actually succeeded
                # this session -- do NOT trust the self-report. Feed a
                # deterministic rejection back into the loop (same
                # mechanism as a malformed-decision retry) instead of
                # ending COMPLETE.
                rejection = (
                    "COMPLETE bi tu choi: ban bao 'mutated: true' nhung chua co mutation "
                    "nao thanh cong duoc ghi nhan trong phien nay (successful_mutation_count=0). "
                    "Neu ban that su can thay doi gi do, hay thuc hien hanh dong that "
                    "(write_file/patch_file/...) truoc; neu day chi la tra loi/phan tich, "
                    "hay complete lai voi 'mutated: false'."
                )
                session.steps.append({
                    "index": len(session.steps) + 1,
                    "decision": decision,
                    "action_result": rejection,
                    "verify": "complete_rejected_no_evidence",
                    "ok": False,
                })
                store.save(session)
                tt.note("complete_rejected_no_evidence")
                continue
            session.status = STATUS_COMPLETE
            session.result_summary = decision["summary"]
            _append_assistant_message(session, decision["summary"])
            store.save(session)
            tt.ret("complete")
            return

        # tool_call
        tool = decision["tool"]
        params = decision.get("params") or {}
        signature = safety.action_signature(tool, params)

        is_dangerous, reason = safety.classify_dangerous(tool, params)
        if is_dangerous:
            already_confirmed = decision["confirmed"] and session.pending_dangerous_signature == signature
            if not already_confirmed:
                session.pending_dangerous_signature = signature
                session.status = STATUS_WAITING
                message = (
                    f"Hanh dong sau can owner xac nhan truoc khi thuc hien: "
                    f"{tool} {params} -- {reason}"
                )
                _append_assistant_message(session, message)
                session.steps.append({
                    "index": len(session.steps) + 1,
                    "decision": decision,
                    "action_result": message,
                    "verify": "dangerous_pending_confirmation",
                    "ok": False,
                })
                store.save(session)
                tt.ret("dangerous_pending_confirmation")
                return
            # Consumed: this exact dangerous action has now been executed
            # once via the confirmed round-trip; a LATER repeat of it must
            # go through confirmation again.
            session.pending_dangerous_signature = None

        workspace_root = Path(session.workspace_root)
        protected_paths = set(session.protected_paths or [])
        pre_run_command_fingerprint = None
        if tool == "run_command":
            pre_run_command_fingerprint = await asyncio.to_thread(_fs_fingerprint, workspace_root)
        action_result_compact: Optional[str] = None
        search_evidence: Optional[Dict[str, Any]] = None
        _tool_t0 = time.monotonic()
        try:
            action_result, replaced_count, action_result_compact, search_evidence = await _execute_action(
                workspace_root, decision, protected_paths
            )
            ok = True
            session.consecutive_failures = 0
            verify = "ok"
            if tool in _UNCONDITIONAL_MUTATION_TOOLS:
                session.successful_mutation_count += 1
            elif tool == "run_command" and pre_run_command_fingerprint is not None:
                # Real mutation evidence, not a blind exit_code trust: only
                # counts when the command actually succeeded AND a bounded
                # before/after filesystem scan of the whole workspace shows
                # a real difference.
                try:
                    parsed_result = json.loads(action_result)
                except (json.JSONDecodeError, TypeError):
                    parsed_result = {}
                command_ok = parsed_result.get("exit_code") == 0 and not parsed_result.get("timed_out")
                if command_ok:
                    post_fingerprint = await asyncio.to_thread(_fs_fingerprint, workspace_root)
                    if post_fingerprint != pre_run_command_fingerprint:
                        session.successful_mutation_count += 1
            if tool == "patch_file" and replaced_count:
                session.total_replacements += replaced_count
                touched_path = params.get("path")
                if touched_path and touched_path not in session.touched_paths:
                    session.touched_paths.append(touched_path)
        except (
            WorkspaceError, safety.DecisionError, FileNotFoundError, ValueError, OSError,
            file_tools.UnsupportedFormatError, file_tools.ProtectedSourceFileError,
            docx_tools.DocxUnavailableError, xlsx_tools.XlsxUnavailableError,
        ) as exc:
            action_result = str(exc)
            ok = False
            session.consecutive_failures += 1
            verify = "action_failed"

        tt.tool(tool, time.monotonic() - _tool_t0, len(str(action_result)), ok)

        if ok:
            prior_index = _find_prior_identical_tool_call(session, signature)
            if prior_index is not None:
                hint = _search_repeat_continuation_hint(search_evidence) if tool == "search_text" else None
                notice = _repeat_notice(prior_index, hint)
                action_result = str(action_result) + notice
                if action_result_compact is not None:
                    action_result_compact = action_result_compact + notice
                tt.note(f"repeat_notice prior_step={prior_index}")

        session.steps.append({
            "index": len(session.steps) + 1,
            "decision": decision,
            "action_result": str(action_result),
            "action_result_compact": action_result_compact,
            "search_evidence": search_evidence,

(a) whether a DECIDE-step JSON is even well-formed enough to execute, and
(b) whether a well-formed tool_call is still dangerous / affects things
outside the workspace enough that it MUST pause for owner confirmation
instead of running automatically. Path-escape itself is a separate,
unconditional reject enforced in workspace.py -- that is a contract
boundary, not a risk judgement this file makes.

Agent-first rewrite (owner directive): this file used to also guess the
OWNER's intent from text -- `is_affirmative_reply` (keyword list) and
`goal_requires_mutation`/`goal_requires_creation` (verb-keyword lists).
Both are deleted. Approval is now a generic action-identity round-trip
(see `classify_dangerous`'s docstring), and false-COMPLETE protection is now

Workbench V2 (owner directive, "kim tay khong kim dau"): also removed the
no-progress/oscillation guard (`MAX_CONSECUTIVE_REPEATED_ACTION`,
`is_oscillating`) that used to pause read-only exploration into WAITING
after a few repeated/alternating reads -- that was Python trying to decide
"the model's strategy looks stuck" and interrupt it, exactly the kind of
cognitive substitution a strong model doesn't need help with. A capable
model reading its own full observation is trusted to change strategy (wider
search, different query, cross-check) on its own; `MAX_STEPS` is now the

meant or whether the model's reasoning is "on track"."""
from __future__ import annotations

import json
import re
from typing import Any, Dict, Tuple

MAX_STEPS = 60
MAX_CONSECUTIVE_MALFORMED = 3
MAX_CONSECUTIVE_FAILURES = 5

DECISION_TYPES = {"reply", "tool_call", "complete"}

KNOWN_TOOLS = {
    "list_tree", "search_text", "read_file",
    "write_file", "patch_file", "make_dir", "delete_path",
    "run_command",
}


def action_signature(tool: str, params: Dict[str, Any]) -> str:
    """Stable identity for 'is this the exact same tool_call as before' --
    tool name + params, normalized so key order never causes a false
    negative. Used ONLY by the dangerous-action reconfirmation mechanic
    (recognizing a re-proposed action after an owner round-trip) -- this
    is identity comparison for a safety gate, not a strategy/no-progress
    heuristic (that guard was removed: a smart model reading its own
    observation is trusted to change strategy on its own; MAX_STEPS above
    is the sole, simple hard ceiling against genuine runaway)."""
    return f"{tool}:{json.dumps(params or {}, ensure_ascii=False, sort_keys=True)}"


_DANGEROUS_COMMAND_PATTERNS = [
    r"\brm\s+-rf\b", r"\bdel\s+/s\b", r"\brd\s+/s\b", r"\bformat\b", r"\bmkfs\b",
    r"\bshutdown\b", r"\breboot\b", r">\s*/dev/sd", r"\bdd\s+if=",
    r"\bgit\s+push\b[^\n]*--force", r"\bgit\s+reset\s+--hard\b",
    r"\bcurl\b[^\n]*\|\s*sh\b", r"\bwget\b[^\n]*\|\s*sh\b",
    r"\bsudo\b", r"\bchmod\s+-R\s+777\b", r":\(\)\s*\{",  # fork-bomb shape
]
_DANGEROUS_COMMAND_RE = re.compile("|".join(_DANGEROUS_COMMAND_PATTERNS), re.IGNORECASE)


class DecisionError(Exception):
    """A DECIDE-step response was malformed or referenced an unknown
    type/tool. Never executed -- always fed back as a VERIFY failure and
    counted against MAX_CONSECUTIVE_MALFORMED."""


def validate_decision(raw: Dict[str, Any]) -> Dict[str, Any]:
    """The only place a raw AI response becomes a trusted 'decision' dict.
    Unknown/malformed input MUST raise DecisionError -- never falls back
    to a guess. Structural validation ONLY (is this shape well-formed,
    does the tool exist) -- never a judgement about whether the CONTENT is
    a good idea; that judgement belongs to the model."""
    if not isinstance(raw, dict):
        raise DecisionError("Decision khong phai JSON object.")
    kind = raw.get("type")
    if not isinstance(kind, str) or kind not in DECISION_TYPES:
        raise DecisionError(f"'type' khong hop le hoac khong xac dinh: {kind!r}")

    if kind == "reply":
        text = raw.get("text")
        if not isinstance(text, str) or not text.strip():
            raise DecisionError("'reply' can truong 'text' khong rong.")
        return {"type": "reply", "text": text}

    if kind == "complete":
        summary = raw.get("summary")
        if not isinstance(summary, str) or not summary.strip():
            raise DecisionError("'complete' can truong 'summary' khong rong.")
        mutated = raw.get("mutated")
        if not isinstance(mutated, bool):
            raise DecisionError("'complete' can truong boolean 'mutated'.")
        return {"type": "complete", "summary": summary, "mutated": mutated}

    # tool_call
    tool = raw.get("tool")
    if not isinstance(tool, str) or tool not in KNOWN_TOOLS:
        raise DecisionError(f"'tool' khong hop le hoac khong xac dinh: {tool!r}")
    params = raw.get("params")
    if params is None:
        params = {}
    if not isinstance(params, dict):
        raise DecisionError("'params' phai la object.")
    confirmed = bool(raw.get("confirmed", False))
    return {"type": "tool_call", "tool": tool, "params": params, "confirmed": confirmed}


def classify_dangerous(tool: str, params: Dict[str, Any]) -> Tuple[bool, str]:
    """Returns (is_dangerous, reason). Purely structural/pattern-based --
    never a guess about intent. Dangerous -> caller must gate on the
    reconfirmation round-trip (see WorkSession.pending_dangerous_signature)
    instead of executing directly."""
    if tool == "delete_path":
        if params.get("recursive"):
            return True, "Xoa de quy 1 thu muc -- can owner xac nhan."
        return False, ""

    if tool == "run_command":
        command = params.get("command")
        command_text = command if isinstance(command, str) else " ".join(str(c) for c in (command or []))
        if _DANGEROUS_COMMAND_RE.search(command_text or ""):
            return True, f"Command bi coi la nguy hiem: {command_text!r}"
        return False, ""

    return False, ""

class _ScriptedInvoke:
    def __init__(self, responses):
        self._responses = list(responses)
        self.calls = []

    async def __call__(self, target, action, payload):
        self.calls.append((target, action, payload))

class _FailIfCalledInvoke:

def test_pagination_reaches_occurrences_past_the_cap_in_the_same_file(tmp_path):

    assert match1["more_occurrences"] is True

    page2 = file_tools.search_text(tmp_path, "needle", "a.txt", match1["next_occurrence_offset"])

        if idx == -1:

def _search_step(index, query, path, workspace_root, start_offset=0, ok=True):
    """Runs the REAL search_text executor (not a hand-built fixture) so
    `search_evidence` is genuine, structurally identical to what
    run_session would actually persist."""
    decision = {"tool": "search_text", "params": {"query": query, "path": path, "start_offset": start_offset}}
    full, _replaced, compact, search_evidence = _execute_sync_action(workspace_root, decision)
    return {
        "index": index,
        "decision": decision,
        "action_result": full,
        "action_result_compact": compact,
        "search_evidence": search_evidence,
        "verify": "ok",
        "ok": ok,
    }

def test_accumulated_evidence_survives_step_compaction(tmp_path):
    # 8 occurrences -> 2 real pages (cap=5/file).
    content = " needle ".join(str(i) for i in range(9))
    (tmp_path / "a.txt").write_text(content, encoding="utf-8")

    step1 = _search_step(1, "needle", "a.txt", tmp_path, start_offset=0)
    next_off = step1["search_evidence"]["files"]["a.txt"]["next_occurrence_offset"]
    step2 = _search_step(2, "needle", "a.txt", tmp_path, start_offset=next_off)

    # Push both search_text steps out of full-detail replay with 2 newer,
    # unrelated tool_calls (mirrors _FULL_DETAIL_TOOL_CALLS=2).
    filler_steps = [
        {
            "index": i, "decision": {"type": "tool_call", "tool": "read_file", "params": {"path": f"f{i}.txt"}},
            "action_result": f"r{i}", "verify": "ok", "ok": True,
        }
        for i in (3, 4)
    ]
    session = _session(steps=[step1, step2] + filler_steps)
    observation = _build_observation(session)

    # Both pages' offsets are still visible via the dedicated chain
    # summary, even though the raw search_text steps themselves are now
    # compacted (or entirely out of the recent-steps full-detail window).
    all_offsets = {m["offset"] for m in step1["search_evidence"]["matches"]} | {
        m["offset"] for m in step2["search_evidence"]["matches"]
    }
    assert len(all_offsets) == 8
    for off in all_offsets:
        assert f"offset {off}" in observation
    assert "Chuoi tim kiem (search_text) da biet" in observation

def _tool_call_step(index, tool, params, ok=True):
    return {
        "index": index,
        "decision": {"type": "tool_call", "tool": tool, "params": params, "confirmed": False},
        "action_result": "some result",
        "verify": "ok" if ok else "action_failed",
        "ok": ok,

class _ScriptedInvoke:
    def __init__(self, responses):
        self._responses = list(responses)
        self.calls = []

    async def __call__(self, target, action, payload):
        self.calls.append((target, action, payload))
        if target == "conversation" and action == "chat":
            text = self._responses.pop(0) if self._responses else json.dumps(
                {"type": "complete", "summary": "done", "mutated": False}
            )
            return HubResponse(success=True, data={"response": text})
        return HubResponse(success=False, error=f"unexpected invoke {target}.{action}")

async def test_successful_retry_after_a_transient_write_failure_carries_no_repeat_notice(tmp_path, monkeypatch):
    workspace = tmp_path / "ws"

    def flaky_write(root, path, content, **kw):
        calls["n"] += 1
        if calls["n"] == 1:
            raise OSError("[WinError 32] The process cannot access the file (transient lock)")
        return real_write(root, path, content, **kw)

    monkeypatch.setattr(agent_loop.file_tools, "write_file", flaky_write)

    same_call = json.dumps({"type": "tool_call", "tool": "write_file",

    await run_session(session.session_id, store, invoke)

    reloaded = store.load(session.session_id)
    assert reloaded.status == STATUS_COMPLETE  # false-COMPLETE gate satisfied by the real retry
    assert reloaded.steps[0]["ok"] is False
    assert reloaded.steps[1]["ok"] is True
    assert (workspace / "out.txt").read_text(encoding="utf-8") == "hello"
    # The successful retry must NOT be told "same as step #1, no new evidence".
    assert "LUU Y THUC TE" not in reloaded.steps[1]["action_result"]
    assert reloaded.successful_mutation_count == 1

@pytest.fixture
def _fresh_trace():
    """trace.py configures its file handler exactly once per process
    (module-level `_CONFIGURED`, same design as Estimation/SonHaiFactory).
    Reset it so a test can bind the handler to its own tmp dir."""
    saved = trace._CONFIGURED
    saved_handlers = list(trace._LOG.handlers)
    trace._CONFIGURED = False
    trace._LOG.handlers.clear()
    try:
        yield
    finally:
        for h in list(trace._LOG.handlers):
            h.close()
        trace._LOG.handlers[:] = saved_handlers
        trace._CONFIGURED = saved
